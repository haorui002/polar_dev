
function segmentTraversal(arr)
    % 功能：遍历长度为2的幂次的数组的特定分段
    % 输入：arr - 长度为2的幂次的数组
    
    n = length(arr);
    
    % 验证数组长度是否为2的幂次
    if n < 2 || bitand(n, n-1) ~= 0
        error('输入数组长度必须是2的幂次');
    end
    
    % 初始化分段参数
    segment_size = n/2;
    current_pos = 1;
    step = 1;
    
    fprintf('开始遍历数组（长度=%d）:\n', n);
    
    while segment_size >= 1
        % 计算当前分段结束位置
        end_pos = min(current_pos + segment_size - 1, n);
        
        % 获取当前分段
        current_segment = arr(current_pos:end_pos);
        
        % 显示当前分段信息
        fprintf('步骤%d: 位置%d-%d [大小=%d]: ', ...
               step, current_pos, end_pos, segment_size);
        disp(current_segment);
        
        % 更新位置和分段大小
        current_pos = end_pos + 1;
        segment_size = segment_size / 2;
        step = step + 1;
        
        % 如果已经到达数组末尾则重置
        if current_pos > n
            current_pos = 1;
            segment_size = n/2;
            fprintf('--- 重置遍历 ---\n');
        end
    end
end
